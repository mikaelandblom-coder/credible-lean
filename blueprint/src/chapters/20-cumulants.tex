\chapter{Extension: moments up to order four}\label{ch:cumulants}

The second-moment core gives the expectation of every quadratic estimator. Its variance
needs moments up to order four. They enter as given tables, like the covariance: the third
cumulant $\kappa_3$ (third central moment) and the fourth cumulant $\kappa_4$ (fourth central
moment minus its three Gaussian pairings). No distribution is assumed; $\kappa_3 = \kappa_4 = 0$
is the Gaussian case. Plan: \texttt{notes/fourth-moments.md}.

\begin{definition}[Variance of a quadratic form]\label{def:varQuad}
  \lean{Credibility.Cumulants.varQuad}
  \leanok
  For a matrix $A$, a covariance $C$, a mean $m$ and tables $\kappa_3$, $\kappa_4$,
  \[ V(A) = 2\operatorname{tr}(ACAC) + 4\,(Am)^\top C (Am)
     + 4\sum_{ijk} A_{ij}(Am)_k\,\kappa_{3,ijk} + \sum_{ijkl} A_{ij}A_{kl}\,\kappa_{4,ijkl}. \]
\end{definition}

\begin{definition}[Fourth cumulants]\label{def:hasFourthCumulant}
  \lean{Credibility.Cumulants.HasFourthCumulant}
  \leanok
  Cells $Z_i$ have fourth cumulants $\kappa_4$ about means $m$, with covariances $C$, if
  $\mathbb E[e_ie_je_ke_l] = \kappa_{4,ijkl} + C_{ij}C_{kl} + C_{ik}C_{jl} + C_{il}C_{jk}$,
  where $e_i = Z_i - m_i$.
\end{definition}

\begin{lemma}[Integrability of products]\label{lem:cumulants_integrable}
  \lean{Credibility.Cumulants.abs_mul4_le, Credibility.Cumulants.integrable_pow4,
        Credibility.Cumulants.integrable_mul4, Credibility.Cumulants.integrable_mul3,
        Credibility.Cumulants.integrable_mul2, Credibility.Cumulants.sum_mul_sum_eq,
        Credibility.Cumulants.integrable_sum_mul_sum, Credibility.Cumulants.integral_sum_mul_sum,
        Credibility.Cumulants.holder442}
  \leanok
  $4|abcd| \le a^4 + b^4 + c^4 + d^4$, so products of up to four $L^4$ functions are integrable on
  a probability space, and the integral of a product of two finite linear combinations is the
  double sum of the integrals of the products.
\end{lemma}
\begin{proof}
  \leanok
  AM--GM twice; then dominated by an integrable function, and linearity.
\end{proof}

\begin{theorem}[The variance of a quadratic estimator]\label{thm:variance_quad_of_cumulants}
  \lean{Credibility.Cumulants.variance_quad_of_cumulants}
  \leanok
  \uses{def:varQuad, def:hasFourthCumulant}
  Let $Z_i$ be random variables in $L^4$ with means $m_i$, covariances $C_{ij}$,
  $\mathbb E[e_ie_je_k] = \kappa_{3,ijk}$ and
  $\mathbb E[e_ie_je_ke_l] = \kappa_{4,ijkl} + C_{ij}C_{kl} + C_{ik}C_{jl} + C_{il}C_{jk}$,
  where $e_i = Z_i - m_i$, and let $A$ be symmetric. Then
  $\operatorname{Var}(\sum_{ij} A_{ij}Z_iZ_j) = V(A)$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{def:varQuad, lem:cumulants_integrable}
  $Z^\top AZ - \mathbb E = U + W$ with $U = \sum A_{ij}(e_ie_j - C_{ij})$ and
  $W = 2(Am)^\top e$. Then $\mathbb E U^2$ is a fourth moment minus $\sum A_{ij}A_{kl}C_{ij}C_{kl}$,
  $\mathbb E UW$ a third moment and $\mathbb E W^2 = 4(Am)^\top C(Am)$; the two remaining pairings
  give $2\operatorname{tr}(ACAC)$ by the symmetry of $A$ and $C$. Products of four $L^4$ functions are
  integrable by $4|abcd| \le a^4 + b^4 + c^4 + d^4$.
\end{proof}

\begin{lemma}[Reordering sums]\label{lem:cumulants_sums}
  \lean{Credibility.Cumulants.sum_comm_last3, Credibility.Cumulants.sum_comm_last4,
        Credibility.Cumulants.diag_transpose_mul_mul}
  \leanok
  A finite sum can be moved from the inside to the outside, and
  $(B^\top AB)_{ss} = \sum_{ij} B_{is}A_{ij}B_{js}$.
\end{lemma}
\begin{proof}
  \leanok
  Exchange two finite sums at a time.
\end{proof}

\begin{theorem}[Independent sources]\label{thm:varQuad_sources}
  \lean{Credibility.Cumulants.varQuad_sources}
  \leanok
  \uses{def:varQuad}
  If the cumulant tables are $\kappa_{3,ijk} = \sum_s c_{3,s}B_{is}B_{js}B_{ks}$ and
  $\kappa_{4,ijkl} = \sum_s c_{4,s}B_{is}B_{js}B_{ks}B_{ls}$ (cells $m + B\xi$ with independent
  sources $\xi_s$ of cumulants $c_{3,s}$, $c_{4,s}$), then with $M = B^\top AB$ and $a = B^\top Am$
  \[ V(A) = 2\operatorname{tr}(ACAC) + 4(Am)^\top C(Am) + 4\sum_s c_{3,s}M_{ss}a_s + \sum_s c_{4,s}M_{ss}^2. \]
  A source drops out of the non-Gaussian part when $M_{ss} = 0$, i.e.\ when the estimator cancels
  its loading.
\end{theorem}
\begin{proof}
  \leanok
  \uses{def:varQuad, lem:cumulants_sums}
  Insert the tables, move the sum over sources to the front, and factor.
\end{proof}

\section{Third moments}

\begin{lemma}[Integrability with three moments]\label{lem:cumulants_integrable3}
  \lean{Credibility.Cumulants.abs_mul3_le, Credibility.Cumulants.integrable_abs_pow3,
        Credibility.Cumulants.integrable_mul3_of_L3, Credibility.Cumulants.integrable_mul2_of_L3}
  \leanok
  $3|abc| \le |a|^3 + |b|^3 + |c|^3$, so products of up to three $L^3$ functions are integrable
  on a probability space.
\end{lemma}
\begin{proof}
  \leanok
  $x^3 + y^3 + z^3 - 3xyz = \tfrac12(x+y+z)\bigl((x-y)^2 + (y-z)^2 + (z-x)^2\bigr)$.
\end{proof}

\begin{definition}[Cubic and mixed moments]\label{def:eCubic}
  \lean{Credibility.Cumulants.eCubic, Credibility.Cumulants.covQuadLin}
  \leanok
  $E_3(T) = \sum_{ijk} T_{ijk}(\kappa_{3,ijk} + m_iC_{jk} + m_jC_{ik} + m_kC_{ij} + m_im_jm_k)$ and
  $V_{2,1}(A, b) = 2(Am)^\top Cb + \sum_{ijk} A_{ij}b_k\kappa_{3,ijk}$.
\end{definition}

\begin{theorem}[Expectation of a cubic estimator]\label{thm:integral_cubic_of_cumulants}
  \lean{Credibility.Cumulants.integral_cubic_of_cumulants}
  \leanok
  \uses{def:eCubic}
  For random cells in $L^3$ with means $m$, covariances $C$ and third central moments
  $\kappa_3$, $\mathbb E\sum_{ijk}T_{ijk}Z_iZ_jZ_k = E_3(T)$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:cumulants_integrable3}
  Write $Z_i = e_i + m_i$, expand each product into eight terms and use $\mathbb E e_i = 0$.
\end{proof}

\begin{theorem}[Covariance of a quadratic and a linear estimator]\label{thm:covariance_quad_lin_of_cumulants}
  \lean{Credibility.Cumulants.covariance_quad_lin_of_cumulants}
  \leanok
  \uses{def:eCubic}
  For random cells in $L^3$ with means $m$, covariances $C$, third central moments $\kappa_3$ and a
  symmetric $A$, $\operatorname{cov}(\sum A_{ij}Z_iZ_j, \sum b_kZ_k) = V_{2,1}(A, b)$. With
  symmetric cells ($\kappa_3 = 0$) only the Gaussian term remains.
\end{theorem}
\begin{proof}
  \leanok
  \uses{def:eCubic, lem:cumulants_integrable, lem:cumulants_integrable3}
  As for the variance: $Z^\top AZ - \mathbb E = U + W$ and $b^\top Z - \mathbb E = b^\top e$; then
  $\mathbb E[Ub^\top e]$ is a third moment and $\mathbb E[Wb^\top e] = 2(Am)^\top Cb$.
\end{proof}

\begin{theorem}[Third moment of residuals]\label{thm:integral_cubic_residuals}
  \lean{Credibility.Cumulants.integral_cubic_residuals}
  \leanok
  Let $X_t = L + e_t$ with any common level $L$ and centred noise in $L^3$ whose third moments
  vanish off the diagonal (e.g.\ independent over years), $\mathbb E e_t^3 = k_t$. For weights
  $\sum a_t = 1$, residuals $r_t = X_t - \sum_s a_sX_s$ and any $c_t$,
  $\mathbb E\sum_t c_tr_t^3 = \sum_t c_t\sum_s(\delta_{ts} - a_s)^3k_s$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:integral_cubic_of_cumulants}
  The level cancels: $r_t = \sum_s(\delta_{ts} - a_s)e_s$, so $\sum c_tr_t^3$ is a cubic form in the
  centred noise. Theorem~\ref{thm:integral_cubic_of_cumulants} with mean $0$ and a diagonal table.
\end{proof}

\begin{theorem}[The app's third-cumulant estimate is unbiased]\label{thm:kappaHat_unbiased}
  \lean{Credibility.Cumulants.kappaHat_unbiased}
  \leanok
  In one region with volumes $w_t > 0$, third cumulant $\kappa/w_t^q$ of the noise,
  $a_t = w_t/\sum w$, the estimate
  $\hat\kappa = \sum_t w_t^qr_t^3 / \sum_t w_t^q\sum_s(\delta_{ts} - a_s)^3w_s^{-q}$
  (\texttt{noiseDiagnostics} in the app) has $\mathbb E\hat\kappa = \kappa$ whenever the denominator is not $0$.
  No distribution is assumed, and nothing about the level.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:integral_cubic_residuals}
  Theorem~\ref{thm:integral_cubic_residuals} with $c_t = w_t^q$, $k_t = \kappa/w_t^q$.
\end{proof}

\begin{theorem}[Fitted variance and fitted mean under skew]\label{thm:covariance_resVar_mean}
  \lean{Credibility.Cumulants.covariance_resVar_mean}
  \leanok
  Cells $X_t = L + e_t$ with any level $L\in L^3$ and centred noise in $L^3$ whose third moments
  vanish off the diagonal, $\mathbb Ee_t^3 = k_t$, and whose squares are uncorrelated with the level.
  For $\sum a_t = 1$, $\hat\mu = \sum_sa_sX_s$, $r_t = X_t - \hat\mu$ and any $c_t$,
  $\operatorname{cov}(\sum_tc_tr_t^2, \hat\mu) = \sum_sa_sk_s\sum_tc_t(\delta_{ts} - a_s)^2$.
\end{theorem}
\begin{proof}
  \leanok
  The level cancels in the residuals, and its part of $\hat\mu - \mathbb E\hat\mu$ meets only
  products $e_te_u$; the noise part gives third moments, diagonal by assumption.
\end{proof}
\begin{remark}[Finding 35 (c)]
  This is the ingredient of the skew channel: under right skew a cohort's fitted variance rises
  with its fitted mean, so weights $\propto 1/\hat\sigma^2$ covary negatively with the estimates
  and the fit under-prices (Finding 35 (a), Theorem~\ref{thm:Core_randomLin_bias}; first order
  through $\partial d_i/\partial\hat\sigma_k^2$).
\end{remark}

\begin{corollary}[The bias of variance-driven weights]\label{cor:randomLin_resVar_bias}
  \lean{Credibility.Core.Estimates.integral_randomLin_affine_sub,
        Credibility.Core.Estimates.integral_randomLin_resVar_sub}
  \leanok
  \uses{thm:Core_randomLin_bias, thm:covariance_resVar_mean}
  Let the weights be $\hat d_i = a_i + b_i\hat\sigma_i^2$ with $\hat\sigma_i^2 \in L^2$ and
  $\sum_i\hat d_i = 1$ in every draw. Then $\mathbb E(\hat d^T\hmu - \mu_j)
  = \sum_ib_i\cov(\hat\sigma_i^2, \hmu_i)$. If each $\hmu_i = \sum_sw_{is}X_{is}$ and
  $\hat\sigma_i^2 = \sum_tc_{it}r_{it}^2$ come from cells as in
  Theorem~\ref{thm:covariance_resVar_mean} (third moments $k_{it}$), the bias is
  $\sum_ib_i\sum_sw_{is}k_{is}\sum_tc_{it}(\delta_{ts} - w_{is})^2$: zero under symmetric noise.
\end{corollary}
\begin{proof}
  \leanok
  \uses{thm:Core_randomLin_bias, thm:covariance_resVar_mean}
  Theorem~\ref{thm:Core_randomLin_bias}; constants drop out of a covariance, so
  $\cov(\hat d_i, \hmu_i) = b_i\cov(\hat\sigma_i^2, \hmu_i)$, which
  Theorem~\ref{thm:covariance_resVar_mean} evaluates.
\end{proof}

\section{Independent sources}

\begin{lemma}[Moments of independent centred sources]\label{lem:moments_indep}
  \lean{Credibility.Cumulants.integral_mul4_indep_of_ne, Credibility.Cumulants.integral_mul3_indep_of_ne,
        Credibility.Cumulants.integral_mul2_indep_of_ne, Credibility.Cumulants.integral_mul4_indep_pair,
        Credibility.Cumulants.integral_mul4_indep, Credibility.Cumulants.integral_mul3_indep,
        Credibility.Cumulants.integral_mul2_indep}
  \leanok
  For independent centred sources $\xi_s$ with variances $v_s$, third moments $c_{3,s}$ and fourth
  moments $c_{4,s} + 3v_s^2$: $\mathbb E\xi_s\xi_t = \delta_{st}v_s$,
  $\mathbb E\xi_s\xi_t\xi_u = \delta_{stu}c_{3,s}$ and
  $\mathbb E\xi_s\xi_t\xi_u\xi_w = \delta_{stuw}c_{4,s} + \delta_{st}\delta_{uw}v_sv_u
  + \delta_{su}\delta_{tw}v_sv_t + \delta_{sw}\delta_{tu}v_sv_t$.
\end{lemma}
\begin{proof}
  \leanok
  A source that appears once is independent of the product of the others, so the expectation
  is $0$. Otherwise the indices are all equal, or form two pairs of different sources, whose
  squares are independent.
\end{proof}

\begin{theorem}[Variance under independent sources]\label{thm:variance_quad_of_sources}
  \lean{Credibility.Cumulants.variance_quad_of_sources}
  \leanok
  \uses{thm:variance_quad_of_cumulants, thm:varQuad_sources}
  Let $Z_i = m_i + \sum_s B_{is}\xi_s$ for independent centred sources $\xi_s$ in $L^4$ (levels,
  shocks, cell noise, or AR(1) innovations), as in Lemma~\ref{lem:moments_indep}, and $A$
  symmetric. With $C = B\operatorname{diag}(v)B^\top$, $M = B^\top AB$ and $a = B^\top Am$,
  \[ \operatorname{Var}\Bigl(\sum A_{ij}Z_iZ_j\Bigr) = 2\operatorname{tr}(ACAC) + 4(Am)^\top C(Am)
     + 4\sum_s c_{3,s}M_{ss}a_s + \sum_s c_{4,s}M_{ss}^2. \]
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:variance_quad_of_cumulants, thm:varQuad_sources, lem:moments_indep, lem:cumulants_integrable}
  By Lemma~\ref{lem:moments_indep}, pushed through $B$, the cells have covariance $C$ and the source
  form of the cumulant tables; then Theorems~\ref{thm:variance_quad_of_cumulants}
  and~\ref{thm:varQuad_sources}.
\end{proof}

\begin{theorem}[Mean under independent sources]\label{thm:integral_quad_of_sources}
  \lean{Credibility.Cumulants.integral_quad_of_sources}
  \leanok
  With the cells of Theorem~\ref{thm:variance_quad_of_sources} and any $A$,
  $\operatorname{E}\sum A_{ij}Z_iZ_j = \sum A_{ij}(m_im_j + C_{ij})$, $C = B\operatorname{diag}(v)B^\top$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:moments_indep, lem:cumulants_integrable}
  Expand $(m_i + e_i)(m_j + e_j)$; the centred cells have mean $0$ and second moments $C$.
\end{proof}

\begin{lemma}[Centering and year-diagonal tables]\label{lem:sampleCov_algebra}
  \lean{Credibility.Cumulants.centering, Credibility.Cumulants.trace_centering_sq,
        Credibility.Cumulants.sum_yearDiag'}
  \leanok
  With $H = I - \mathbf 1\mathbf 1^\top/T$, $\operatorname{tr}H^2 = T - 1$; a table that vanishes
  off the year diagonal sums over one year index.
\end{lemma}
\begin{proof}
  \leanok
  Direct computation.
\end{proof}


\begin{theorem}[Variance of a sum of products of linear statistics]\label{thm:variance_sum_mul_lin}
  \lean{Credibility.Cumulants.variance_sum_mul_lin}
  \leanok
  For cells $e_p$ in $L^4$ with second moments $C$ and fourth moments
  $\kappa_{4,pqrs} + C_{pq}C_{rs} + C_{pr}C_{qs} + C_{ps}C_{qr}$, and vectors $u_t$, $v_t$,
  \[ \operatorname{Var}\sum_t(u_t^\top e)(v_t^\top e) = \sum_{t,t'}\bigl[(u_t^\top Cu_{t'})(v_t^\top Cv_{t'})
     + (u_t^\top Cv_{t'})(v_t^\top Cu_{t'}) + \kappa_4(u_t, v_t, u_{t'}, v_{t'})\bigr]. \]
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:cumulants_integrable}
  $\operatorname{E}Q^2 - (\operatorname{E}Q)^2$, each term a fourth moment of the cells; the pairing
  $C_{pq}C_{rs}$ cancels against $(\operatorname{E}Q)^2$.
\end{proof}

\begin{definition}[Residual coefficients and year-diagonal covariance]\label{def:resCross}
  \lean{Credibility.Cumulants.resCross, Credibility.Cumulants.resCoef, Credibility.Cumulants.yearCov}
  \leanok
  The residual $r_{ct} = X_{ct} - \sum_sa_{cs}X_{cs}$ has coefficients $\delta_{ts} - a_{cs}$ on
  cohort $c$; $N\otimes$ years is the covariance $N_s$ in year $s$ and none across years;
  $G^{bc}_{tt'} = \sum_s(\delta_{ts} - a_{bs})(\delta_{t's} - a_{cs})N_{s,bc}$.
\end{definition}

\begin{lemma}[Covariance of two residuals]\label{lem:resCoef_dot}
  \lean{Credibility.Cumulants.resCoef_dot}
  \leanok
  \uses{def:resCross}
  The residual coefficient vectors give $\operatorname{cov}(r_{bt}, r_{ct'}) = G^{bc}_{tt'}$.
\end{lemma}
\begin{proof}
  \leanok
  Only cohort $b$, cohort $c$ and equal years contribute.
\end{proof}

\begin{theorem}[Variance of the shock-covariance estimator]\label{thm:variance_crossResid}
  \lean{Credibility.Cumulants.variance_crossResid}
  \leanok
  \uses{def:resCross}
  Cells $X_{ct} = L_c + e_{ct}$ with any random levels and noise in $L^4$, uncorrelated across
  years with covariance $N_s$ in year $s$, whose fourth cumulants vanish unless all four cells are
  in the same year ($k_s$ there). For weights $\sum_ta_{ct} = 1$ and any $D$,
  \[ \operatorname{Var}\frac{\sum_tr_{it}r_{jt}}{D} = \frac{1}{D^2}\Bigl(\sum_{t,t'}\bigl[G^{ii}_{tt'}G^{jj}_{tt'}
     + G^{ij}_{tt'}G^{ji}_{tt'}\bigr] + \sum_s(1 - a_{is} - a_{js} + Ta_{is}a_{js})^2k_{s,ijij}\Bigr). \]
  With $D = \sum_td_{ijt}$ this is the variance of $\hat C_{ij}$ in the shock model.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:variance_sum_mul_lin, lem:resCoef_dot, lem:sampleCov_algebra}
  The levels cancel in the residuals; Theorem~\ref{thm:variance_sum_mul_lin} with the residual
  coefficients, and the cumulant term sums over one year.
\end{proof}

\begin{theorem}[Variance of a sample covariance]\label{thm:variance_sampleCov}
  \lean{Credibility.Cumulants.variance_sampleCov}
  \leanok
  Cells $Z_{at}$ in $L^4$ with means $\mu_a$, covariance $V\otimes I$ and fourth cumulants that
  vanish unless all four cells are in the same year, where they are $k_{abde}$ (e.g.\ independent
  years). For $T\ge 2$ and
  $\hat s_{ij} = \sum_t(Z_{it} - \bar Z_i)(Z_{jt} - \bar Z_j)/(T-1)$,
  \[ \operatorname{Var}\hat s_{ij} = \frac{V_{ii}V_{jj} + V_{ij}^2}{T-1} + \frac{k_{ijij}}{T}, \]
  also for $i = j$. The first term is the normal-theory variance; skewness does not enter.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:variance_crossResid, lem:sampleCov_algebra}
  Theorem~\ref{thm:variance_crossResid} with equal weights $1/T$ and $D = T - 1$: then
  $G^{bc}_{tt'} = H_{tt'}V_{bc}$, $\sum_{tt'}H_{tt'}^2 = T - 1$, and each year's cumulant weight is
  $(1 - 1/T)^2$.
\end{proof}

\section{The statistics of the pooled fits: covariances (S53)}

The pooled fits (Bühlmann--Straub, corrBS, the BLUP on corrBS's $\hat K$) depend on the data
only through three quadratic statistics: the pooled within-region variance $P$, the
cross-region residual products $Q_n$ and the spread of the region means $B$. Their first-order
price needs their covariances.

\begin{definition}[Covariance of two quadratic forms]\label{def:covQuad}
  \lean{Credibility.Cumulants.covQuad}
  \leanok
  \uses{def:varQuad}
  $\mathrm{cov}(A, B) = 2\operatorname{tr}(ACBC) + 4(Am)^\top C(Bm)
  + 2\sum A_{ij}(Bm)_k\kappa_{3,ijk} + 2\sum B_{ij}(Am)_k\kappa_{3,ijk}
  + \sum A_{ij}B_{kl}\kappa_{4,ijkl}$.
\end{definition}

\begin{lemma}[Polarization]\label{lem:varQuad_add}
  \lean{Credibility.Cumulants.varQuad_add, Credibility.Cumulants.covQuad_self}
  \leanok
  \uses{def:covQuad}
  $\mathrm{cov}(A, A) = V(A)$. For symmetric $C$ and $\kappa_{4,klij} = \kappa_{4,ijkl}$,
  $V(A + B) = V(A) + 2\,\mathrm{cov}(A, B) + V(B)$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{def:covQuad}
  Expand; $\operatorname{tr}(BCAC) = \operatorname{tr}(ACBC)$ by cyclicity.
\end{proof}

\begin{lemma}[Quadratic forms are square integrable]\label{lem:Cumulants_memLp_quadForm}
  \lean{Credibility.Cumulants.memLp_quadForm}
  \leanok
  A quadratic form in cells in $L^4$ is in $L^2$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:cumulants_integrable}
  Hölder with exponents $4, 4, 2$.
\end{proof}

\begin{theorem}[The covariance of two quadratic estimators]\label{thm:covariance_quad_of_cumulants}
  \lean{Credibility.Cumulants.covariance_quad_of_cumulants}
  \leanok
  \uses{def:covQuad}
  Under the hypotheses of Theorem~\ref{thm:variance_quad_of_cumulants}, for symmetric $A$, $B$:
  $\cov(Z^\top AZ, Z^\top BZ) = \mathrm{cov}(A, B)$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:variance_quad_of_cumulants, lem:varQuad_add, lem:Cumulants_memLp_quadForm}
  $\Var(X + Y) = \Var X + 2\cov(X, Y) + \Var Y$ with $X + Y = Z^\top(A + B)Z$, and
  Lemma~\ref{lem:varQuad_add}; $C$ and $\kappa_4$ are symmetric as moments.
\end{proof}

\begin{lemma}[Sums of products of linear statistics]\label{lem:symQuad}
  \lean{Credibility.Cumulants.symQuad, Credibility.Cumulants.symQuad_symm,
        Credibility.Cumulants.sum_symQuad, Credibility.Cumulants.symQuad_mulVec}
  \leanok
  $\sum_k c_k(u_k^\top x)(v_k^\top x) = x^\top Ax$ with the symmetric
  $A = \sum_k c_k(u_kv_k^\top + v_ku_k^\top)/2$, and
  $Ay = \sum_k (c_k/2)((v_k^\top y)u_k + (u_k^\top y)v_k)$.
\end{lemma}
\begin{proof}
  \leanok
  Exchange finite sums.
\end{proof}

\begin{definition}[The statistics $P$, $Q_n$, $B$]\label{def:fitStats}

\end{definition}

\begin{theorem}[Exact variances and covariances of $P$, $Q_n$, $B$]\label{thm:covariance_stat}

\begin{proof}

\begin{lemma}[When the region means and the residuals are uncorrelated]\label{lem:meanResid_uncorr}
  \lean{Credibility.Cumulants.resCoef_dotProduct_regionConst,
        Credibility.Cumulants.mulVec_resCoef,
        Credibility.Cumulants.meanCoef_dotProduct_resCoef_eq_zero}
  \leanok
  \uses{def:fitStats}
  With $\sum_t a_{jt} = 1$, residuals annihilate region-constant vectors. If
  $C_{(it),(ju)} = \Lambda_{ij} + \delta_{tu}D_{t,ij}$ (a level covariance plus year noise) and
  $a_{it}D_{t,ij}$ does not depend on $t$, then $\cov(\bar x_i, r_{jt}) = 0$. This holds for
  uncorrelated regions with $D_{t,ii} = \sigma_i^2/w_{it}$, and for
  $D_{t,ij} = V_{ij}/\sqrt{w_{it}w_{jt}}$ with proportional volumes $w_{it} = W_iv_t$.
\end{lemma}
\begin{proof}
  \leanok
  $(Cr_{jt})_{(i,u)} = D_{u,ij}(\delta_{tu} - a_{ju})$, so
  $\cov(\bar x_i, r_{jt}) = c_{ij} - c_{ij}\sum_u a_{ju} = 0$.
\end{proof}

\begin{theorem}[Means and residuals covary only through $\kappa_4$]\label{thm:covariance_meanStat_resStat}
  \lean{Credibility.Cumulants.covariance_meanStat_resStat}
  \leanok
  \uses{def:fitStats, lem:symQuad}
  Cells with region-constant means, no third cumulants and region means uncorrelated with the
  residuals. Then a statistic $S$ of the region means and a statistic $R$ of the residuals
  (sums of products) have $\cov(S, R) = \sum A_{S,ij}A_{R,kl}\kappa_{4,ijkl}$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:covariance_symQuad, lem:symQuad, lem:fitStats_eval, lem:meanResid_uncorr}
  $A_Rm = 0$, because residuals annihilate region-constant means; this removes the mean term and one
  skewness term, and $\kappa_3 = 0$ removes the other. $A_RCy$ is uncorrelated with every region
  mean, so $A_SCA_R = 0$ and the trace term vanishes.
\end{proof}

\begin{theorem}[$\cov(B, P) = \cov(B, Q_n) = 0$]\label{thm:covariance_statB}
  \lean{Credibility.Cumulants.covariance_statB_eq, Credibility.Cumulants.covariance_statB_eq_zero}
  \leanok
  \uses{thm:covariance_stat}

  cumulants vanish too (e.g.\ normal cells).
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:covariance_meanStat_resStat}
  Theorem~\ref{thm:covariance_meanStat_resStat} with $S = B$ and $R = P$ or $Q_n$.
\end{proof}

\begin{theorem}[The balanced identity]\label{thm:bsZ_balanced}

\begin{lemma}[First-order variance of $\hat z$]\label{lem:zHatVar}
  \lean{Credibility.Cumulants.zHatVar, Credibility.Cumulants.zHatVar_eq}
  \leanok
  The delta-method variance of $\hat z = 1 - \hat v/S^2$ at $(v, s)$ is
  $s^{-2}\Var\hat v - 2vs^{-3}\cov(\hat v, S^2) + v^2s^{-4}\Var S^2$; with
  $\cov(\hat v, S^2) = 0$ (Theorem~\ref{thm:covariance_statB}) it is
  $(1 - z)^2(\mathrm{CV}^2(\hat v) + \mathrm{CV}^2(S^2))$, $z = 1 - v/s$.
\end{lemma}
\begin{proof}
  \leanok
  Algebra.
\end{proof}

\begin{theorem}[Law (a): the first-order price of fitting Bühlmann--Straub]\label{thm:bsPrice_balanced}
  \lean{Credibility.Cumulants.bsPrice_of_balanced, Credibility.Cumulants.bsPrice_le_of_balanced}
  \leanok
  \uses{lem:zHatVar}

  (Theorem~\ref{thm:variance_sampleCov} with $\kappa_4 = 0$). The curvature $(1 - 1/J)s$ of the
  blend's error times the first-order variance of $\hat z$, relative to $(1 - z)\tau^2$, is
  \[ \frac2J\Bigl(1 + \frac{J - 1}{J(T - 1)}\Bigr)\frac{1 - z}{z} \le \frac2J\,\frac T{T - 1}\,\frac{1 - z}z. \]
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:zHatVar}
  Lemma~\ref{lem:zHatVar}; $(1 - 1/J)\cdot 2/(J - 1) = 2/J$ and $(J - 1)/J \le 1$.
\end{proof}

\begin{lemma}[First-order price of an own factor]\label{lem:ownPrice}
  \lean{Credibility.Cumulants.zOwnVar, Credibility.Cumulants.zOwnVar_eq,
        Credibility.Cumulants.ownPrice_eq}
  \leanok
  For a region's own factor $\hat z = \tau^2/(\tau^2 + \hat v)$ ($\tau^2$ fixed), the delta-method
  variance at $v$ is $(\tau^2/s^2)^2\Var\hat v = z^2(1 - z)^2\,\mathrm{CV}^2(\hat v)$
  ($\partial z/\partial\log v = -z(1 - z)$). With $\mathrm{CV}^2(\hat v) = c$, the curvature
  $(1 - 1/J)s$ times it, relative to $(1 - z)\tau^2$, is $(1 - 1/J)\,z(1 - z)\,c$.
\end{lemma}
\begin{proof}
  \leanok
  Algebra, with $1 - z = v/s$ and $sz = \tau^2$.
\end{proof}

\begin{theorem}[Law (b): the price of own variances]\label{thm:ownPrice_balanced}
  \lean{Credibility.Cumulants.ownPrice_sub_pooled_of_balanced,
        Credibility.Cumulants.ownPrice_sub_pooled_le_of_balanced}
  \leanok
  \uses{lem:ownPrice}
  Balanced world, $J$ regions, $T$ years. With the normal-theory $\mathrm{CV}^2 = 2/(T - 1)$ of a
  region's own $\hat v_j$ and $2/(J(T - 1))$ of one pooled $\hat v$, the extra first-order price of
  own factors over pooled ones, relative to $(1 - z)\tau^2$, is
  \[ \Bigl(1 - \frac1J\Bigr)^2 z(1 - z)\,\frac2{T - 1} \;\le\; z(1 - z)\,\frac2{T - 1}
     \;\le\; \frac1{2(T - 1)}. \]
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:ownPrice}
  Lemma~\ref{lem:ownPrice} with $c = 2/(T - 1)$ minus $c = 2/(J(T - 1))$:
  $(1 - 1/J)(1 - 1/J)$. Then $(1 - 1/J)^2 \le 1$ and $z(1 - z) \le 1/4$.
\end{proof}
\begin{corollary}[An own factor against Bühlmann--Straub's]\label{cor:ownPrice_vs_bsPrice}
  \lean{Credibility.Cumulants.ownPrice_eq_sq_mul_bsPrice}
  \leanok

\end{corollary}
\begin{proof}
  \leanok

  $\partial z/\partial\log v$ is $-z(1 - z)$ against $-(1 - z)$.
\end{proof}

  Equal volumes in Theorem~\ref{thm:crossTerm_bs}: all $z_i = z$, $\alpha = ze_j + (1 - z)\mathbf 1/J$,
  $\lambda = -z(1 - z)(e_j - \mathbf 1/J)$ and $\mathrm{zHatLin} = -z(1 - z)\hat\varepsilon$, so the integrands agree.
  With $D$ known, only $S^2$ moves: Lemma~\ref{lem:integral_errW} for $B$ and
  Lemma~\ref{lem:integral_lin_lin_sumSq} for $\operatorname{E}[A(\bar x_j - \bar x)] = 0$.
\end{proof}

\begin{remark}[Finding 38: estimating $D$ flips the noise part]
  With $D$ known the result is Chen, Lahiri and Rao's $2g_4$ (up to $1 - 1/J$): heavy-tailed levels
  lower the MSE, heavy-tailed noise raises it. With $D$ estimated from the same cells the noise part
  changes sign and matches the level part. Added to the kurtosis terms of law (a)
  (Theorem~\ref{thm:bsPrice_balanced_cumulants}), the total first-order effect of kurtosis is
  $-(1 - 1/J)z(1 - z)(\gamma_L + \gamma_N/T)/J$: heavy tails of either kind lower the first-order MSE
  of the plug-in.
\end{remark}

\begin{definition}[Any volumes: primitives, Bühlmann--Straub's weights and their direction]\label{def:primCellsW}
  \lean{Credibility.Cumulants.primVarW, Credibility.Cumulants.diagK4, Credibility.Cumulants.primK4W, Credibility.Cumulants.linW, Credibility.Cumulants.errW,

\end{definition}

\begin{lemma}[Bühlmann--Straub's weights and their derivative]\label{lem:bsAlpha}

\begin{lemma}[The error and a spread against $P$ and $B$, any volumes]\label{lem:integral_errW}

\begin{theorem}[The first-order cross term, any volumes]\label{thm:crossTerm_bs}
  \lean{Credibility.Cumulants.crossTerm_bs_of_cumulants}
  \leanok

     - \frac{T_B}{c\tau^2}\Bigr), \]

  Theorem~\ref{thm:crossTerm_balanced_cumulants}.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:integral_errW, lem:bsAlpha, lem:integral_lin_lin_sumSq}

  Lemma~\ref{lem:integral_errW} the two moments.
\end{proof}

\begin{remark}[Finding 39: the noise part has no sign]
  Region 0 holds $8/9$ of its volume in one year, so its mean is nearly one cell: its noise kurtosis

\begin{theorem}[The level part of the cross term is $\le 0$, any volumes]\label{thm:crossTerm_bs_level_nonpos}
  \lean{Credibility.Cumulants.sum_mul_sum_sq_le_of_monovary, Credibility.Cumulants.bsZk_spread_monovary, Credibility.Cumulants.bsLevel_sum_nonneg, Credibility.Cumulants.crossTerm_bs_level_nonpos}
  \leanok
  \uses{thm:crossTerm_bs}
  With $\kappa_L \ge 0$ and $\kappa_N = 0$ the cross term of Theorem~\ref{thm:crossTerm_bs} is $\le 0$ for any
  positive volumes: $\sum_i \lambda_i (1 - W_i/W) W_i (\alpha_i - \delta_{ij}) \ge 0$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:crossTerm_bs}
  With $m_i = W_i(1 - W_i/W)$, $Z = \sum z$, $R = \sum_{i \ne j} z_i$, $s' = \sum_{i \ne j} z_i^2$, the sum is
  $(1 - z_j)^2/Z^3$ times
  $m_j z_j R (R^2 + s') + z_j \sum_{i \ne j} m_i z_i^2 (R + z_i) + \bigl(R \sum_{i \ne j} m_i z_i^3 - s' \sum_{i \ne j} m_i z_i^2\bigr)$.
  The last bracket is Chebyshev's sum inequality with weights $z_i$: $m_i z_i$ increases with $z_i$, since
  $W_i \le W_l$ and $W_i + W_l \le W$ give $m_i \le m_l$. And $c = W - \sum W_i^2/W \ge 0$.
\end{proof}

\begin{remark}[Finding 40: not for the BLUP on corrBS's $\hat K$]
  For any fit that sees the data only through $(P, Q_n, B)$ the level part is
  $(\kappa_L/c)\, \tfrac{d}{d\tau^2} \sum_p m_p (d_{jp} - \delta_{pj})^2$: more $\hat\tau^2$ must pull the weights towards
  the own region. For the plug-in BLUP this fails: $J = 3$, $T = 2$, $\rho = 0.99$, $\tau^2 = 1/20$, $\sigma^2 = 1$,
  volumes $(0.2^2, 0.23^2)$, $(0.13^2, 0.14^2)$, $(35^2, 44^2)$, region 1 gives $+0.12\%$ of its MSE per unit $\gamma_L$
  (exact rational check). In the worlds searched it stays below $+0.4\%$, and needs strong correlation and
  very uneven volumes.
\end{remark}

\begin{remark}[Finding 38 for any volumes]
  Every $\lambda_i$, $\alpha_i$ and weight in Theorem~\ref{thm:crossTerm_bs} is explicit; on unbalanced worlds
  (region sizes $16\times$ apart) the term is negative in every region, for levels and for noise, and
  matches simulation (\texttt{studies/runs/cross-term.js}); with one region's volume in a single year the
  noise part turns positive (Theorem~\ref{thm:crossTerm_bs_pos_example}).
\end{remark}

\begin{lemma}[The gap between two blends is a quadratic in $z$]\label{lem:Core_linMse_lineWeight}
  \lean{Credibility.Core.linMse_lineWeight, Credibility.Core.linMse_lineWeight_sub}
  \leanok
  \uses{def:Core_blup}
  For symmetric $K$, any collective weights $b$ and $c = e_j - b$:
  $\mathrm{mse}_j(ze_j + (1 - z)b) = \mathrm{mse}_j(b) + 2z\,c^\top(Kb - \ell) + z^2c^\top Kc$.
  So the gap between two blends with collective weights $b$, $b'$ and the same $z$ is
  $\Delta_0 + 2z\Delta_1 + z^2\Delta_2$ with $\Delta_0 = \mathrm{mse}_j(b) - \mathrm{mse}_j(b')$,
  $\Delta_1 = c^\top(Kb - \ell) - c'^\top(Kb' - \ell)$, $\Delta_2 = c^\top Kc - c'^\top Kc'$; it is
  $0$ at $z = 1$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:Core_linMse_pythagoras}
  $ze_j + (1 - z)b = b + zc$; expand and use $b^\top Kc = c^\top Kb$.
\end{proof}

\section{The price of the BLUP over the blend on the same $\hat K$}

The plug-in BLUP and the plug-in blend read the same fitted $\hat K$, but their weights move
differently when $\hat K$ errs. Prasad and Rao's $g_3$ in the general form of Das, Jiang and Rao
(2004, eq.~(3.4)), $\operatorname{tr}\{(\partial d)^\top K(\partial d)\,\Var\hat\theta\}$, is the
expectation of the squared weight error below over the fit's error $\hat\theta - \theta$. These
results give that error exactly for both families, and its first-order limit. Distribution-free:
they are statements about weights.

\begin{definition}[The zero-sum solve]\label{def:Core_zeroSumSolve}
  \lean{Credibility.Core.zeroSumSolve}
  \leanok
  \uses{def:glsWeight}
  $Qr = K^{-1}r - (1^\top K^{-1}r)\,b$, with the GLS weights $b$.
\end{definition}

\begin{lemma}[$Qr$ is the zero-sum solution]\label{lem:Core_zeroSumSolve}
  \lean{Credibility.Core.sum_zeroSumSolve, Credibility.Core.mulVec_zeroSumSolve,
        Credibility.Core.dotProduct_of_mulVec_sub_const, Credibility.Core.quad_of_mulVec_sub_const,
        Credibility.Core.zeroSumSolve_quad, Credibility.Core.eq_zeroSumSolve,
        Credibility.Core.zeroSumSolve_smul}
  \leanok
  \uses{def:Core_zeroSumSolve}
  For positive definite $K$: $\sum_i (Qr)_i = 0$ and $KQr - r$ is constant; $Qr$ is the only
  zero-sum vector with that property; $(Qr)^\top K(Qr) = r^\top Qr$; $Q(tr) = tQr$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:sum_glsWeight, lem:mulVec_glsWeight}
  $Kb = s^2 1$ and $\sum_i b_i = 1$. For zero-sum $u$ and $K\delta - r$ constant,
  $u^\top K\delta = u^\top r$; a difference of two solutions $e$ then has $e^\top Ke = 0$.
\end{proof}

\begin{theorem}[The BLUP's weight error on a fitted $K'$]\label{thm:Core_blupWeight_sub}
  \lean{Credibility.Core.blupWeight_sub_eq, Credibility.Core.glsWeight_sub_eq,
        Credibility.Core.blupWeight_sub_quad}
  \leanok
  \uses{def:Core_blup, lem:Core_zeroSumSolve}
  For positive definite $K$, $K'$ and any $L$, $L'$, with $Q' = Q$ on $K'$:
  \[ d^*(K', L') - d^*(K, L) = Q'\bigl(L'_j - L_j - (K' - K)d^*\bigr), \qquad
     b(K') - b(K) = -Q'\bigl((K' - K)b\bigr), \]
  and the first has $K'$-norm $r^\top Q'r$, $r = L'_j - L_j - (K' - K)d^*$. Exact.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:Core_zeroSumSolve, lem:Core_blup_foc}
  The difference has zero sum and $K'(d^{*\prime} - d^*) - r = (K'd^{*\prime} - L'_j) - (Kd^* - L_j)$,
  a difference of constants (Lemma~\ref{lem:Core_blup_foc}); uniqueness of $Q'r$.
\end{proof}

\begin{theorem}[The blend's weight error on a fitted $K'$]\label{thm:Core_blendWeight_sub}
  \lean{Credibility.Core.blendWeight_sub_eq, Credibility.Core.blendDir_quad,
        Credibility.Core.blendWeight_sub_quad}
  \leanok
  \uses{def:Core_blup, lem:Core_zeroSumSolve}
  With $K'$ and a factor $z'$ in place of $K$, $z$:
  $w' - w = (z' - z)(e_j - b') - (1 - z)\,Q'\bigl((K' - K)b\bigr)$, whose $K'$-norm is
  \[ (z' - z)^2\,\mathrm{gap}' - 2(z' - z)(1 - z)\,(e_j - b')^\top(K' - K)b
     + (1 - z)^2\,y^\top Q'y, \qquad y = (K' - K)b, \]
  with $\mathrm{gap}' = K'_{jj} - s'^2$. Exact.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:Core_blupWeight_sub, lem:Core_quadForm_gap, lem:Core_zeroSumSolve}
  The GLS part of Theorem~\ref{thm:Core_blupWeight_sub}. In the norm, $e_j - b'$ has zero sum and
  $K'Q'y - y$ is constant, so the cross term is $(e_j - b')^\top y$;
  Lemma~\ref{lem:Core_quadForm_gap} gives the gap.
\end{proof}

\begin{definition}[First-order prices]\label{def:Core_blupPrice}
  \lean{Credibility.Core.blupPrice, Credibility.Core.blendPrice}
  \leanok
  \uses{def:Core_blup, def:Core_zeroSumSolve}
  In the direction $(\Delta K, \Delta L)$, with the factor moving at rate $\Delta z$:
  \[ \mathrm{price}_{\mathrm{BLUP}} = r^\top Qr,\ r = \Delta L_j - \Delta K\,d^*; \qquad
     \mathrm{price}_{\mathrm{blend}} = \Delta z^2\,\mathrm{gap}
       - 2\Delta z(1 - z)(e_j - b)^\top\Delta K\,b + (1 - z)^2\,(\Delta K b)^\top Q(\Delta K b). \]
\end{definition}

\begin{lemma}[Continuity along a line]\label{lem:Core_price_continuity}
  \lean{Credibility.Core.continuous_mulVec_left, Credibility.Core.continuousAt_inv_line,
        Credibility.Core.continuousAt_gls_zeroSumSolve_line, Credibility.Core.tendsto_quad,
        Credibility.Core.quad_smul_div}
  \leanok
  \uses{def:Core_zeroSumSolve}
  For positive definite $K$, $t \mapsto (K + t\Delta K)^{-1}$, the GLS weights and $Qy$ on
  $K + t\Delta K$ are continuous at $0$.
\end{lemma}
\begin{proof}
  \leanok
  Continuity of the matrix inverse at an invertible matrix, and of $x \mapsto 1/x$ at
  $1^\top K^{-1}1 > 0$.
\end{proof}

\begin{theorem}[The prices at first order]\label{thm:Core_tendsto_price}
  \lean{Credibility.Core.tendsto_blupPrice, Credibility.Core.tendsto_blendPrice}
  \leanok
  \uses{def:Core_blupPrice}
  Along $K + t\Delta K$ (positive definite near $t = 0$) and $L + t\Delta L$, with a factor $z(t)$,
  $z'(0) = \Delta z$, the squared weight errors in the true $K$-norm, divided by $t^2$, tend to
  $\mathrm{price}_{\mathrm{BLUP}}$ and $\mathrm{price}_{\mathrm{blend}}$ as $t \to 0$. By
  Lemma~\ref{lem:Core_linMse_pythagoras} each squared weight error is the extra MSE of those
  weights over the best ones, so $\mathrm{price}_{\mathrm{BLUP}} - \mathrm{price}_{\mathrm{blend}}$
  is the first-order extra price of the BLUP's weights over the blend's.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:Core_blupWeight_sub, thm:Core_blendWeight_sub, lem:Core_price_continuity,
        lem:Core_zeroSumSolve}
  Along the line $r$ is exactly $t$ times its direction, so the BLUP's error is $t\,Q_t r$
  (Theorem~\ref{thm:Core_blupWeight_sub}) and the blend's $t$ times
  $\frac{z(t) - z}{t}(e_j - b_t) - (1 - z)Q_t(\Delta K b)$
  (Theorem~\ref{thm:Core_blendWeight_sub}); let $t \to 0$
  (Lemma~\ref{lem:Core_price_continuity}) and use $(Qr)^\top KQr = r^\top Qr$ and the norm of
  Theorem~\ref{thm:Core_blendWeight_sub} at $K' = K$.
\end{proof}

\begin{remark}[Finding 45: neither price dominates]
  The BLUP reacts to $\Delta K$ through the whole vector $\Delta K\,d^*$; the blend only through
  $\Delta z$ and $\Delta K\,b$. Neither is always cheaper: for random positive definite $K$ and
  symmetric $\Delta K$ the BLUP's price is the lower one in 11.7\% of directions (study S73), so the
  sign of the extra price depends on $\Var\hat\theta$. Its size on corrBS's $\hat K$ (S73, $J = 8$,
  $T = 5$): $+32$ to $+38$ points of the floor at app volume with $\sigma_j$ spread 2 and
  $\rho = 0.3$--$0.6$, against a gain of 3--12; at most $+4$ with equal $\sigma_j$. So
  \texttt{Engine.gainPrice}'s ``none extra'' is wrong where study S72 saw the BLUP lose.
\end{remark}

\section{Under a variance function (distribution layer)}

Under a variance function the noise cumulants depend on the cell's own mean, which is random:
the noise is uncorrelated with the level but not independent of it. The levels are a random
point $a\sim\mu$; given them the noise has the law $\kappa(a)$. The joint law is $\mu\otimes\kappa$.
This section assumes a noise family (\texttt{VarianceFunction.lean}); nothing in the core uses it.

\begin{theorem}[Law of total variance]\label{thm:variance_compProd}
  \lean{Credibility.VarianceFunction.variance_compProd}
  \leanok
  For $f\in L^2(\mu\otimes\kappa)$ with every section $f(a,\cdot)\in L^2(\kappa(a))$,
  $\operatorname{Var} f = \operatorname{E}_a\operatorname{Var}(f\mid a) + \operatorname{Var}_a\operatorname{E}(f\mid a)$.
\end{theorem}
\begin{proof}
  \leanok
  $\operatorname{Var}f = \operatorname{E}f^2 - (\operatorname{E}f)^2$, integrate in two steps, and
  $\operatorname{E}(f\mid a)^2 \le \operatorname{E}(f^2\mid a)$ makes the conditional mean square integrable.
\end{proof}

\begin{lemma}[Quadratic forms of $L^4$ cells]\label{lem:memLp_quad}
  \lean{Credibility.VarianceFunction.memLp_quad}
  \leanok
  A quadratic form in $L^4$ cells is in $L^2$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{lem:Cumulants_memLp_quadForm}
  Lemma~\ref{lem:Cumulants_memLp_quadForm} for the cells $m + B\xi$.
\end{proof}

\begin{theorem}[Variance with level-dependent noise cumulants]\label{thm:variance_quad_of_conditional_sources}
  \lean{Credibility.VarianceFunction.variance_quad_of_conditional_sources}
  \leanok
  Given the levels $a$, let the cells be $Z = m(a) + B\xi$ with sources independent and centred
  under $\kappa(a)$, with variances $v_s(a)$, third moments $c_{3,s}(a)$ and fourth cumulants
  $c_{4,s}(a)$, and let the estimator be in $L^2(\mu\otimes\kappa)$. With $C(a) = B\operatorname{diag}(v(a))B^\top$
  and $M = B^\top AB$, $A$ symmetric,
  \[ \operatorname{Var}\Bigl(\sum A_{ij}Z_iZ_j\Bigr) = \operatorname{E}_a\Bigl[2\operatorname{tr}(ACAC) + 4(Am)^\top C(Am)
     + 4\sum_s c_{3,s}M_{ss}(B^\top Am)_s + \sum_s c_{4,s}M_{ss}^2\Bigr]
     + \operatorname{Var}_a\Bigl(\sum A_{ij}(m_im_j + C_{ij})\Bigr). \]
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:variance_compProd, lem:memLp_quad, thm:variance_quad_of_sources, thm:integral_quad_of_sources}
  The law of total variance (cumulance); given $a$, Theorems~\ref{thm:variance_quad_of_sources}
  and~\ref{thm:integral_quad_of_sources} under $\kappa(a)$.
\end{proof}

\begin{definition}[Cumulants of the variance function $\mu^p$]\label{def:tweedie}
  \lean{Credibility.VarianceFunction.tweedieVar, Credibility.VarianceFunction.tweedieκ₃,
        Credibility.VarianceFunction.tweedieκ₄}
  \leanok
  For dispersion $\varphi/w$ and mean $\mu$: $V = \tfrac{\varphi}{w}\mu^p$,
  $\kappa_3 = (\tfrac{\varphi}{w})^2p\,\mu^{2p-1}$, $\kappa_4 = (\tfrac{\varphi}{w})^3p(2p-1)\,\mu^{3p-2}$,
  from $\kappa_{n+1} = \tfrac{\varphi}{w}V(\mu)\,d\kappa_n/d\mu$ (Tweedie). $p = 0$: normal; $p = 1$: Poisson-like.
\end{definition}

\begin{lemma}[Trace with a diagonal covariance]\label{lem:trace_mul_diagonal}
  \lean{Credibility.VarianceFunction.trace_mul_diagonal_mul_diagonal}
  \leanok
  For symmetric $A$, $\operatorname{tr}(ADAD) = \sum_{ij}A_{ij}^2d_id_j$ with $D = \operatorname{diag}(d)$.
\end{lemma}
\begin{proof}
  \leanok
  Expand the trace.
\end{proof}

\begin{theorem}[Variance under the variance function $\mu^p$]\label{thm:variance_quad_of_powerVariance}
  \lean{Credibility.VarianceFunction.variance_quad_of_powerVariance}
  \leanok
  \uses{def:tweedie, thm:variance_quad_of_conditional_sources}
  Given the levels, let the cell noises be independent and centred with the cumulants of
  Definition~\ref{def:tweedie} at their own mean $m_i(a)$, and $A$ symmetric. Then
  \[ \operatorname{Var}\Bigl(\sum A_{ij}Z_iZ_j\Bigr) = \operatorname{E}_a\Bigl[2\sum_{ij}A_{ij}^2V_iV_j
     + 4\sum_i V_i(Am)_i^2 + 4\sum_i\kappa_{3,i}A_{ii}(Am)_i + \sum_i\kappa_{4,i}A_{ii}^2\Bigr]
     + \operatorname{Var}_a\Bigl(m^\top Am + \sum_i A_{ii}V_i\Bigr). \]
  The cumulants are averaged over the levels exactly, not taken at the mean level.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:variance_quad_of_conditional_sources, lem:trace_mul_diagonal}
  Theorem~\ref{thm:variance_quad_of_conditional_sources} with the cells as sources ($B = 1$).
\end{proof}
