\chapter{The balance property}\label{ch:balance}
% Thesis §2.9. Lean: Credibility/Balance.lean

Blended estimates are \emph{balanced} if
$\sum_{j,t} w_{jt}\mu_{jt} = \sum_{j,t} w_{jt}\hmu_j^b$: had the insurer
charged $\hmu_j^b$ historically, the net result would be zero. With
volume weights $a_{jt} = w_{jt}/w_{j*}$ this reads
$\sum_j w_{j*}(1 - z_j)(\hmu_j - \hmu) = 0$. It fails for the proposed model
in general (the simulation study measures by how much), but holds for
Bühlmann--Straub.

Balance is a property of the blend alone, for any factors $z_j$ and weights
$b_j$ with $\sum_j b_j = 1$.

\begin{definition}[Balanced blend]\label{def:Core_balanced}
  \lean{Credibility.Core.Balanced}
  \leanok
  The blend $\hmu_j^b = z_j\hmu_j + (1 - z_j)\sum_i b_i\hmu_i$ is
  \emph{balanced for the weights $u$} if $\sum_j u_j(\hmu_j - \hmu_j^b) = 0$
  for all numbers $\hmu_j$.
\end{definition}

\begin{theorem}[Balance of any blend]\label{thm:Core_balance}
  \lean{Credibility.Core.sum_mul_sub_blend, Credibility.Core.balanced_iff}
  \leanok
  \uses{def:Core_balanced}
  Let $\sum_j b_j = 1$. Then
  $\hmu_j - \hmu_j^b = (1 - z_j)(\hmu_j - \hmu)$ with $\hmu = \sum_i b_i\hmu_i$,
  and the blend is balanced for the weights $u$ if and only if
  $u_j(1 - z_j) = c\,b_j$ for some constant $c$.
\end{theorem}
\begin{proof}
  \leanok
  With $a_j = u_j(1 - z_j)$ and $a_* = \sum_j a_j$,
  $\sum_j u_j(\hmu_j - \hmu_j^b) = \sum_j (a_j - a_* b_j)\hmu_j$. This vanishes
  for all $\hmu$ iff every coefficient does ($\hmu$ a unit vector), i.e.\
  $a_j = a_* b_j$. Conversely, $a_j = c\,b_j$ gives $a_* = c$.
\end{proof}

\begin{corollary}[Balance with diagonal $K$]\label{cor:Core_balance_diag}
  \lean{Credibility.Core.balanced_iff_of_diag}
  \leanok
  \uses{thm:Core_balance, thm:Core_diag}
  Let $K = \operatorname{diag}(\tau^2 + v_j)$ with positive entries, $v_j \ne 0$
  and $J \ge 2$, and take the optimal $z_j$ and GLS weights $b_j$. Then the
  blend is balanced for the weights $u$ if and only if
  $u_j \propto 1/(K_{jj} - \tau^2) = 1/v_j$.
\end{corollary}
\begin{proof}
  \leanok
  \uses{thm:Core_balance, lem:sum_glsWeight, thm:Core_diag}
  By Theorem~\ref{thm:Core_diag}, $1 - z_j = v_j/(\tau^2 + v_j)$ and
  $b_j = (\tau^2 + v_j)^{-1}/V$, so the condition of
  Theorem~\ref{thm:Core_balance} says that $u_j v_j$ is constant.
\end{proof}

So the blend is balanced for the \emph{precisions} $1/v_j$ of the cohort
noise; with $v_j = \sigma^2/w_{j*}$ these are proportional to the volumes,
which is Bühlmann--Straub.

\begin{remark}[Study findings S27--S28: balance is the wrong target under correlation]\label{rem:balance_correlation}
  \uses{thm:Core_balance, cor:Core_balance_diag}
  By Theorem~\ref{thm:Core_balance} the optimal blend is balanced, for all
  data, exactly for weights $u_j \propto b_j/(1 - z_j)$. In Bühlmann--Straub
  these are the volumes $w_{j*}$; with a diagonal $K$ they are the precisions
  $w_{j*}/\sigma_j^2$, so with region-specific $\sigma_j$ the blend is not
  exposure-balanced. Under correlation some $b_j$ are negative, so no
  positive volume measure is balanced. A simulation
  (\texttt{studies/LOG.md}) finds negative $b_j$ in every world with
  $\rho > 0$ even with the true parameters, and an exposure imbalance above
  1\% in most draws when the $\sigma_j$ differ. Forcing exposure balance
  afterwards (shifting or scaling the rates, or a Bühlmann--Straub-style
  collective) costs up to about 30\% more error at $\rho = 0.6$: the
  exposure-weighted mean carries the common year shock, which the GLS
  weights hedge away and balance puts back into every rate. At $\rho = 0$
  with equal $\sigma$ it costs nothing.
\end{remark}

\begin{definition}[Balanced factors]\label{def:Core_balancedZ}
  \lean{Credibility.Core.balancedZ, Credibility.Core.bestBalancedC}
  \leanok
  For weights $u_j \ne 0$, $b_j$ and a number $c$, let
  $z_j(c) = 1 - c\,b_j/u_j$. For loss weights $\varphi_j$, per-cohort moments
  with $g_j = \Var(\hmu_j - \hmu)$ and optimal factors $z_j^*$, and
  $a_j = b_j/u_j$, let
  $c^* = \sum_j\varphi_j g_j(1 - z_j^*)a_j \big/ \sum_j\varphi_j g_j a_j^2$.
\end{definition}

\begin{lemma}[The balanced blends form a line]\label{lem:Core_balanced_line}
  \lean{Credibility.Core.balanced_iff_balancedZ}
  \leanok
  \uses{thm:Core_balance, def:Core_balancedZ}
  Let $\sum_j b_j = 1$ and $u_j \ne 0$. The blend is balanced for $u$ if and
  only if $z = z(c)$ for some $c$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{thm:Core_balance}
  Divide $u_j(1 - z_j) = c\,b_j$ by $u_j$.
\end{proof}

\begin{theorem}[The best balanced blend]\label{thm:Core_bestBalanced}
  \lean{Credibility.Core.sum_errVar_balancedZ, Credibility.Core.mul_bestBalancedC,
        Credibility.Core.bestBalanced_optimal}
  \leanok
  \uses{lem:Core_balanced_line, lem:errVar_eq}
  Let $\sum_j b_j = 1$, $u_j \ne 0$, $\varphi_j \ge 0$ and $g_j > 0$. Then
  $z(c^*)$ is balanced for $u$, and among all blends balanced for $u$ it
  minimises $\sum_j\varphi_j\,\mathrm{MSE}_j(z_j)$, for any (correlated)
  second moments.
\end{theorem}
\begin{proof}
  \leanok
  \uses{lem:Core_balanced_line, lem:errVar_eq}
  By Lemma~\ref{lem:errVar_eq},
  $\sum_j\varphi_j\mathrm{MSE}_j(z_j(c)) = \sum_j\varphi_j\mathrm{MSE}_j(z_j^*) + S_0 - 2cS_1 + c^2S_2$
  with $S_0 = \sum\varphi g(1 - z^*)^2$, $S_1 = \sum\varphi g(1 - z^*)a$,
  $S_2 = \sum\varphi g a^2 \ge 0$. Always $S_2c^* = S_1$ (if $S_2 = 0$ every
  term of $S_1$ vanishes), so the excess over $c^*$ is $S_2(c - c^*)^2 \ge 0$.
\end{proof}

\begin{corollary}[The cost of balance]\label{cor:Core_bestBalanced_cost}
  \lean{Credibility.Core.bestBalanced_cost}
  \leanok
  \uses{thm:Core_bestBalanced}
  If $g_j \ne 0$ and $S_2 \ne 0$, the best balanced blend loses
  $S_0 - S_1^2/S_2 \ge 0$ against the optimal blend.
\end{corollary}
\begin{proof}
  \leanok
  \uses{thm:Core_bestBalanced}
  Put $c = S_1/S_2$ in the expansion above.
\end{proof}

\begin{remark}[Finding 15: balance by a shift, not by bending $z$]\label{rem:bestBalanced}
  \uses{thm:Core_bestBalanced}
  The line $z(c)$ fixes the shape $1 - z_j \propto b_j/u_j$: large cohorts
  are pushed to $z_j \approx 1$, small ones far from $z_j^*$. In the app's
  worlds the best balanced blend has up to about twice the error of the
  optimal blend (at most $+96\%$), while adding one common shift to all
  rates (the benchmarking of Wang, Fuller and Qu, 2008, which leaves the
  blend class) costs at most $41\%$. Both are linear rules with the true
  $K$, so these numbers are computed exactly from the second moments, not
  simulated (\texttt{studies/runs/exact-balance.js}).
\end{remark}

\begin{definition}[Imbalance and corrections]\label{def:Core_imbalance}
  \lean{Credibility.Core.imbalanceWeight, Credibility.Core.imbalanceCov,
        Credibility.Core.correctionLoss, Credibility.Core.bestCorrectionC}
  \leanok
  For a blend with factors $z$ and weights $b$, the imbalance
  $D = \sum_j u_j(\hmu_j - \hmu_j^b) = \sum_i w_i\hmu_i$ has weights
  $w_i = u_i(1 - z_i) - \bigl(\sum_j u_j(1 - z_j)\bigr)b_i$. A correction is
  $\theta_j = \hmu_j^b + c_jD$. Let $\kappa_j = z_j(Kw)_j - (Lw)_j$ (it will be
  $\cov(\hmu_j^b - \mu_j, D)$) and $V = w^\top K w = \Var D$. For loss weights
  $\varphi_j$ the loss is
  $\sum_j\varphi_j(\mathrm{MSE}_j + 2c_j\kappa_j + c_j^2V)$, and
  $c_j^* = (\lambda u_j/\varphi_j - \kappa_j)/V$ with
  $\lambda = (V + \sum_i u_i\kappa_i)/\sum_i(u_i^2/\varphi_i)$.
\end{definition}

\begin{lemma}[The imbalance is linear]\label{lem:Core_imbalance}
  \lean{Credibility.Core.sum_mul_sub_blend_eq, Credibility.Core.sum_imbalanceWeight}
  \leanok
  \uses{def:Core_imbalance, thm:Core_balance}
  $\sum_j u_j(\hmu_j - \hmu_j^b) = \sum_i w_i\hmu_i$ for all data, and if
  $\sum_j b_j = 1$ then $\sum_i w_i = 0$, so $\mathbb E D = 0$.
\end{lemma}
\begin{proof}
  \leanok
  \uses{thm:Core_balance}
  Expand $\hmu_j - \hmu_j^b = (1 - z_j)(\hmu_j - \hmu)$.
\end{proof}

\begin{theorem}[The best balanced correction (Wang--Fuller--Qu)]\label{thm:Core_bestCorrection}
  \lean{Credibility.Core.sum_mul_bestCorrectionC, Credibility.Core.bestCorrection_optimal}
  \leanok
  \uses{def:Core_imbalance}
  Let $\varphi_j > 0$, $V > 0$ and some $u_j \ne 0$. Then
  $\sum_j u_jc_j^* = 1$ (so $\theta^*$ is balanced), and among all balanced
  corrections ($\sum_j u_jc_j = 1$) $c^*$ minimises the loss.
\end{theorem}
\begin{proof}
  \leanok
  Each term is a parabola in $c_j$ and $\varphi_j(\kappa_j + Vc_j^*) = \lambda u_j$. So
  the loss of $c$ is the loss of $c^*$ plus $V\sum_j\varphi_j(c_j - c_j^*)^2$ plus
  $2\lambda\sum_j u_j(c_j - c_j^*) = 0$.
\end{proof}

\begin{corollary}[The common shift]\label{cor:Core_commonShift}
  \lean{Credibility.Core.commonShift_optimal}
  \leanok
  \uses{thm:Core_bestCorrection}
  If $\varphi = u$, $u_j > 0$, $V > 0$ and $\kappa_j = k$ for all $j$, the
  common shift $c_j = 1/\sum_i u_i$, i.e.\ $\theta_j = \hmu_j^b + D/\sum_iu_i$, is
  the best balanced correction.
\end{corollary}
\begin{proof}
  \leanok
  \uses{thm:Core_bestCorrection}
  Then $\lambda = (V + k\sum u)/\sum u$ and $c_j^* = (\lambda - k)/V = 1/\sum_i u_i$.
\end{proof}

\begin{lemma}[Independent cohorts: $\kappa = 0$]\label{lem:Core_imbalanceCov_diag}
  \lean{Credibility.Core.imbalanceCov_of_diag}
  \leanok
  \uses{def:Core_imbalance, thm:Core_diag}
  For $K = \mathrm{diag}(\tau^2 + v_j) \succ 0$, $L = \tau^2I$, $|J| \ge 2$ and the
  optimal factors $z_j^*$, $\kappa_j = 0$ for every $w$: the blend's error is
  uncorrelated with every imbalance.
\end{lemma}
\begin{proof}
  \leanok
  \uses{thm:Core_diag}
  $z_j^* = \tau^2/(\tau^2 + v_j)$, so $z_j^*(\tau^2 + v_j)w_j - \tau^2w_j = 0$.
\end{proof}

\begin{remark}[Finding 16: the shift is optimal only without correlation]\label{rem:commonShift}
  \uses{cor:Core_commonShift}
  Under correlation $\kappa_j$ varies with $j$ and the shift is not the best
  correction: with three cohorts, $\tau^2 = 1$, $v = (0.5, 1, 3)$,
  $\rho = 0.5$, $L = \tau^2I$, $u = (3, 2, 1)$, $c^* = (0.177, 0.121, 0.227)$
  against $1/6$ and the loss is $5.237$ against $5.249$. Over random worlds
  ($\rho$ up to $0.95$) the shift is never more than about $2\%$ worse, and
  $c^*$ needs $\kappa$, so the shift stays the practical choice.
\end{remark}

\begin{theorem}[Bühlmann--Straub is balanced]\label{thm:bs_balance}
  \lean{Credibility.bs_balance}
  \leanok
  \uses{def:bsZ}
  Let $W_j > 0$, $\kappa > 0$, $z_j = W_j/(W_j + \kappa)$ and
  $\hmu = \sum_j z_j\hmu_j / \sum_j z_j$, with $J \ge 1$. Then for
  \emph{any} numbers $\hmu_j$,
  $\sum_j W_j(1 - z_j)(\hmu_j - \hmu) = 0$.
\end{theorem}
\begin{proof}
  \leanok
  \uses{thm:Core_balance}
  With $b_j = z_j/z_*$ we have $\sum_i b_i\hmu_i = \hmu$ and
  $W_j(1 - z_j) = W_j\kappa/(W_j + \kappa) = \kappa z_j = \kappa z_*\,b_j$, so
  Theorem~\ref{thm:Core_balance} applies with $u = W$.
\end{proof}

This holds whether or not the model assumptions \eqth{28} are met: it is
an identity in the data.
